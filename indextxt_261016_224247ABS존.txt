<script>
const video = document.getElementById('video');
const canvas = document.getElementById('canvas');
const ctx = canvas.getContext('2d', { willReadFrequently: true });

// 상태 변수
let facingMode = 'environment'; // 기본 후면 카메라
let zoomLevel = 1.0;
let zoneOffsetX = -65; // 타자 중심에서 홈플레이트까지의 가로 오프셋
let useROI = true;
let autoBallDetect = true;

// 볼카운트 및 투구 기록
let balls = 0, strikes = 0, outs = 0;
let pitchHistory = []; // {x, y, r, isStrike, clipped, num}
let lastPitchTime = 0;

// ABS 스트라이크 존 기본 좌표 (타자 미인식 시 화면 중앙 기준, 인식 시 실시간 업데이트)
let absZone = { x: 200, y: 180, w: 110, h: 170 };
let batterLandmarks = null;
let prevFrameData = null;

// FPS 측정
let frameCount = 0, lastFpsTime = performance.now();

// 1. 음성 합성 (AI 로봇 심판 콜)
function speakCall(text) {
if (!('speechSynthesis' in window)) return;
window.speechSynthesis.cancel();
const utter = new SpeechSynthesisUtterance(text);
utter.lang = 'ko-KR';
utter.rate = 1.15;
utter.pitch = 1.0;
window.speechSynthesis.speak(utter);
}

// 2. 전광판 업데이트
function updateScoreboard() {
document.querySelectorAll('.dot.b').forEach((el, i) => el.classList.toggle('active', i < balls));
document.querySelectorAll('.dot.s').forEach((el, i) => el.classList.toggle('active', i < strikes));
document.querySelectorAll('.dot.o').forEach((el, i) => el.classList.toggle('active', i < outs));
}

// 3. 핵심: KBO ABS 걸침 판정 함수 (공의 반지름이 존 경계에 1픽셀이라도 걸치면 스트라이크)
function judgePitch(bx, by, br) {
const now = performance.now();
if (now - lastPitchTime < 900) return; // 중복 판정 방지 (0.9초 쿨타임)
lastPitchTime = now;

const Xmin = absZone.x;
const Xmax = absZone.x + absZone.w;
const Ytop = absZone.y;
const Ybot = absZone.y + absZone.h;

// 원(야구공)의 중심과 사각형(ABS 존) 사이의 최단 거리 계산
const dx = Math.max(Xmin - bx, 0, bx - Xmax);
const dy = Math.max(Ytop - by, 0, by - Ybot);
const distance = Math.sqrt(dx * dx + dy * dy);

// 공의 반지름(br) 이내로 경계선에 닿았으면 STRIKE!
const isStrike = distance <= br;
const isCenterInside = (bx >= Xmin && bx <= Xmax && by >= Ytop && by <= Ybot);
const isClipped = isStrike && !isCenterInside; // 중심은 밖인데 공 끝이 걸친 보더라인 스트라이크

const pitchNum = pitchHistory.length + 1;
pitchHistory.push({ x: bx, y: by, r: br, isStrike, isClipped, num: pitchNum });

const banner = document.getElementById('callBanner');
const title = document.getElementById('callTitle');
const sub = document.getElementById('callSub');

banner.className = 'call-banner show ' + (isStrike ? 'strike' : 'ball');

if (isStrike) {
title.innerText = 'STRIKE!';
sub.innerText = isClipped ? 🔥 보더라인 걸침 판정! (공 끝 ${(br - distance).toFixed(1)}px 걸침): '한가운데 통과 (스트라이크)'; speakCall(isClipped ? '스트라이크! 보더라인 걸쳤습니다!' : '스트라이크!'); strikes++; if (strikes &gt;= 3) { setTimeout(() =&gt; speakCall('삼진 아웃!'), 1100); strikes = 0; balls = 0; outs = (outs + 1) % 3; } } else { title.innerText = 'BALL!'; sub.innerText =존에서 ${(distance - br).toFixed(1)}px 빠짐;
speakCall('볼!');
balls++;
if (balls >= 4) {
setTimeout(() => speakCall('볼넷! 진루하세요!'), 1100);
balls = 0; strikes = 0;
}
}

updateScoreboard();
setTimeout(() => { banner.classList.remove('show'); }, 1800);
}

// 4. MediaPipe Pose 설정 (타자 체형 인식 -> KBO ABS 56.35% / 27.64% 계산)
const pose = new Pose({
locateFile: (file) => https://cdn.jsdelivr.net/npm/@mediapipe/pose/${file}`
});
pose.setOptions({
modelComplexity: 0, // 모바일 렉 방지를 위해 초경량 고속 모델(0) 사용
smoothLandmarks: true,
minDetectionConfidence: 0.5,
minTrackingConfidence: 0.5
});

pose.onResults((results) => {
if (results.poseLandmarks) {
batterLandmarks = results.poseLandmarks;
document.getElementById('batterStatus').innerText = '✅ 타자 체형 락온 완료';
} else {
batterLandmarks = null;
document.getElementById('batterStatus').innerText = '고정 존 모드 (타자 탐색 중)';
}
});

// 5. 실시간 고속 공(흰색 빠른 물체) 감지 엔진 (ROI 내부만 스캔하여 렉 방지!)
function detectMovingBallInROI() {
if (!autoBallDetect) return;

// 볼 판정까지 포함한 외곽 감지 구역 (ROI)
const pad = 65;
const rx = Math.max(0, Math.floor(absZone.x - pad));
const ry = Math.max(0, Math.floor(absZone.y - pad));
const rw = Math.min(canvas.width - rx, Math.floor(absZone.w + pad * 2));
const rh = Math.min(canvas.height - ry, Math.floor(absZone.h + pad * 2));

if (rw <= 0 || rh <= 0) return;

const imgData = ctx.getImageData(rx, ry, rw, rh);
const data = imgData.data;

if (prevFrameData && prevFrameData.length === data.length) {
let sumX = 0, sumY = 0, count = 0;

// 4픽셀 간격으로 고속 스캔 (움직임 변화량 + 밝은 야구공 색상 동시 충족 픽셀 탐색)
for (let y = 0; y < rh; y += 4) {
for (let x = 0; x < rw; x += 4) {
const i = (y * rw + x) * 4;
const r = data[i], g = data[i+1], b = data[i+2];
const pr = prevFrameData[i], pg = prevFrameData[i+1], pb = prevFrameData[i+2];

const diff = Math.abs(r - pr) + Math.abs(g - pg) + Math.abs(b - pb);
const brightness = (r + g + b) / 3;

// 빠르게 움직이면서(diff > 110) 밝은 물체(brightness > 165)인 경우 야구공으로 인식
if (diff > 110 && brightness > 165) {
sumX += x;
sumY += y;
count++;
}
}
}

// 야구공 크기에 해당하는 움직임 클러스터가 잡히면 즉시 판정!
if (count >= 12 && count <= 220) {
const ballX = rx + (sumX / count);
const ballY = ry + (sumY / count);
const ballRadius = Math.max(12, Math.min(24, Math.sqrt(count) * 2.2));
judgePitch(ballX, ballY, ballRadius);
}
}
prevFrameData = new Uint8ClampedArray(data);
}

// 6. 메인 렌더링 루프
let lastPoseRun = 0;
async function renderLoop(now) {
frameCount++;
if (now - lastFpsTime >= 1000) {
document.getElementById('fpsText').innerText = frameCount;
frameCount = 0;
lastFpsTime = now;
}

if (video.readyState >= 2) {
if (canvas.width !== video.videoWidth) {
canvas.width = video.videoWidth;
canvas.height = video.videoHeight;
}

// 1) 카메라 원본 그리기
ctx.drawImage(video, 0, 0, canvas.width, canvas.height);

// 2) 0.1초마다 타자 체형 인식 비동기 실행 (프레임 드랍 방지)
if (now - lastPoseRun > 100) {
lastPoseRun = now;
pose.send({ image: video }).catch(() => {});
}

// 3) KBO ABS 스트라이크 존 위치 계산
if (batterLandmarks) {
const nose = batterLandmarks[0];
const lShoulder = batterLandmarks[11], rShoulder = batterLandmarks[12];
const lHip = batterLandmarks[23], rHip = batterLandmarks[24];
const lKnee = batterLandmarks[25], rKnee = batterLandmarks[26];
const lAnkle = batterLandmarks[27], rAnkle = batterLandmarks[28];

// 타자 관절 시각화 (초록색 스켈레톤 포인트)
[lShoulder, rShoulder, lHip, rHip, lKnee, rKnee].forEach(pt => {
if (pt && pt.visibility > 0.4) {
ctx.beginPath();
ctx.arc(pt.x * canvas.width, pt.y * canvas.height, 6, 0, Math.PI * 2);
ctx.fillStyle = '#22c55e';
ctx.fill();
}
});

// 타자의 전체 키(머리~발목) 추정 후 KBO 공식 비율(56.35% ~ 27.64%) 적용
const shoulderY = ((lShoulder.y + rShoulder.y) / 2) * canvas.height;
const kneeY = ((lKnee.y + rKnee.y) / 2) * canvas.height;
const centerX = ((lShoulder.x + rShoulder.x) / 2) * canvas.width;
const shoulderWidth = Math.abs(lShoulder.x - rShoulder.x) * canvas.width;

// 발목까지 다 보이면 신장 공식 그대로 적용, 상반신 위주면 어깨~무릎 비례 적용
if (lAnkle && lAnkle.visibility > 0.5 && nose && nose.visibility > 0.5) {
const headTopY = (nose.y * canvas.height) - (shoulderWidth * 0.4);
const ankleY = ((lAnkle.y + rAnkle.y) / 2) * canvas.height;
const fullHeight = Math.max(100, ankleY - headTopY);

absZone.y = ankleY - (fullHeight * 0.5635); // 상단 56.35%
const botY = ankleY - (fullHeight * 0.2764); // 하단 27.64%
absZone.h = Math.max(60, botY - absZone.y);
} else {
// 어깨와 무릎 기준 KBO ABS 보정
absZone.y = shoulderY + (kneeY - shoulderY) * 0.18;
absZone.h = Math.max(70, (kneeY - shoulderY) * 0.78);
}
absZone.w = Math.max(65, shoulderWidth * 0.95); // 홈플레이트 47.18cm 비례 폭
absZone.x = centerX + zoneOffsetX - (absZone.w / 2);
} else {
// 타자가 화면에 없을 때는 화면 중앙에 기본 ABS 존 표시
absZone.w = canvas.width * 0.22;
absZone.h = canvas.height * 0.32;
absZone.x = (canvas.width - absZone.w) / 2;
absZone.y = (canvas.height - absZone.h) / 2;
}

// 4) ROI 내부 움직이는 공 감지 실행
detectMovingBallInROI();

// 5) ROI(관심 영역) 가이드 박스 그리기
if (useROI) {
const pad = 65;
ctx.strokeStyle = 'rgba(239, 68, 68, 0.6)';
ctx.lineWidth = 2;
ctx.setLineDash([6, 6]);
ctx.strokeRect(absZone.x - pad, absZone.y - pad, absZone.w + pad * 2, absZone.h + pad * 2);
ctx.setLineDash([]);
ctx.fillStyle = 'rgba(239, 68, 68, 0.85)';
ctx.font = 'bold 11px sans-serif';
ctx.fillText('⚡ ROI 고속 스캔 구역 (볼 판정 경계)', absZone.x - pad + 4, absZone.y - pad - 6);
}

// 6) KBO ABS 9분할 스트라이크 존 그리기
ctx.strokeStyle = '#38bdf8';
ctx.lineWidth = 4;
ctx.strokeRect(absZone.x, absZone.y, absZone.w, absZone.h);
ctx.fillStyle = 'rgba(56, 189, 248, 0.12)';
ctx.fillRect(absZone.x, absZone.y, absZone.w, absZone.h);

// 3x3 내부 격자
ctx.strokeStyle = 'rgba(56, 189, 248, 0.45)';
ctx.lineWidth = 1.5;
for (let i = 1; i <= 2; i++) {
ctx.beginPath();
ctx.moveTo(absZone.x + (absZone.w / 3) * i, absZone.y);
ctx.lineTo(absZone.x + (absZone.w / 3) * i, absZone.y + absZone.h);
ctx.stroke();

ctx.beginPath();
ctx.moveTo(absZone.x, absZone.y + (absZone.h / 3) * i);
ctx.lineTo(absZone.x + absZone.w, absZone.y + (absZone.h / 3) * i);
ctx.stroke();
}

// 7) 투구 탄착군(야구공 궤적) 표시
pitchHistory.forEach(p => {
ctx.beginPath();
ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
ctx.fillStyle = p.isStrike ? '#eab308' : '#22c55e';
ctx.fill();
ctx.lineWidth = 2.5;
ctx.strokeStyle = '#fff';
ctx.stroke();

ctx.fillStyle = '#000';
ctx.font = 'bold 12px sans-serif';
ctx.textAlign = 'center';
ctx.textBaseline = 'middle';
ctx.fillText(p.num, p.x, p.y);
});
ctx.textAlign = 'start';
}

requestAnimationFrame(renderLoop);
}

// 7. 화면 터치/클릭 시 즉시 투구 판정 테스트 (영상 촬영 시 100% 성공 보장용!)
canvas.addEventListener('pointerdown', (e) => {
const rect = canvas.getBoundingClientRect();
// 줌 배율 보정 좌표 계산
const scaleX = canvas.width / rect.width;
const scaleY = canvas.height / rect.height;
const clickX = (e.clientX - rect.left) * scaleX;
const clickY = (e.clientY - rect.top) * scaleY;

const ballRadius = Math.max(14, absZone.w * 0.085); // 실제 야구공 비율 반지름
judgePitch(clickX, clickY, ballRadius);
});

// 8. 카메라 시작 함수
async function startCamera() {
if (video.srcObject) {
video.srcObject.getTracks().forEach(t => t.stop());
}
try {
const stream = await navigator.mediaDevices.getUserMedia({
